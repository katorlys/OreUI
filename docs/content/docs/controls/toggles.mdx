```tsx
import { TabButton, Toggles } from "@oreui-web/react/toggles";
import { useState } from "react";

export function DaylightToggles() {
  const [value, setValue] = useState("day");

  return (
    <Toggles aria-label="Daylight cycle">
      <TabButton selected={value === "day"} onChange={() => setValue("day")}>
        Day
      </TabButton>
      <TabButton
        selected={value === "night"}
        onChange={() => setValue("night")}
      >
        Night
      </TabButton>
      <TabButton
        selected={value === "always"}
        onChange={() => setValue("always")}
      >
        Always
      </TabButton>
    </Toggles>
  );
}
```

import { ref } from "vue";

    <TabButton :selected="value === 'day'" @change="value = 'day'"

    >
    <TabButton :selected="value === 'night'" @change="value = 'night'"
      >Night</TabButton
    >
    <TabButton :selected="value === 'always'" @change="value = 'always'"

    >
  </Toggles>
</template>

## Behavior

Toggles sets `role="tablist"`, lays out its children, and applies the `toggle` palette to direct and nested Tab Buttons.  
For more information, see [Tab Button](../../buttons/tab-button/).

## Accessibility

Give the tab group an accessible name. `Arrow keys` move through enabled options. `Home` and `End` select the first and last option.
