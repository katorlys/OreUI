## Usage

<Tabs groupId="framework" items={["Vanilla", "React", "Astro"]} persist>



    </Tab>
    <Tab value="React">

```tsx
import { Textfield } from "@katorlys/oreui-react/textfield";
import { useState } from "react";

export function WorldNameField() {
  const [value, setValue] = useState("");

  return (
    <Textfield
      name="world"
      label="World name"
      description="Shown to other players"
      placeholder="New World"
      value={value}
      onInput={(event) => {
        const field = event.target as HTMLElement & { value: string };
        setValue(field.value);
      }}
    />
  );
}
```

## Validation

Use native input constraints or set `error` for a custom validation message.

```html
<ore-textfield label="Seed" type="number" required></ore-textfield>
<ore-textfield
  label="World name"
  error="This name is already in use"
></ore-textfield>
```

## Events

Textfield dispatches `input` whenever the value changes. Its public `value` property stays synchronized with the internal input.

## Accessibility

Textfield connects its label, description, and validation message to the native input. Use a visible `label` first, and use `aria-label` only when visible label is not appropriate.
