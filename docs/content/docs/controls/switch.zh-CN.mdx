import { Switch } from "@oreui-web/react/switch";
import { useState } from "react";

export function AutosaveSwitch() {
  const [checked, setChecked] = useState(false);

  return (
    <Switch
      checked={checked}
      name="autosave"
      value="enabled"
      variant="icons"
      onChange={(event) => {
        setChecked(event.currentTarget.checked);
      }}
    >
      Enable autosave
    </Switch>
  );
}

## 表单

选中项的值会包含在 `FormData` 中，重置表单会恢复初始状态。

## 事件

原生 input 的 checked 状态发生变化后，会触发 `input` 和 `change` 事件。

## 属性

Switch 使用原生 checkbox 的 `checked`、`disabled`、`name`、`required` 和 `value` 属性。使用 `ore-switch` class 和 `role="switch"` 提供开关样式与语义。OreUI 选项使用 `data-color` 和 `data-variant="icons"`。

## CSS 变量

| 变量                                   | 用途           |
| -------------------------------------- | -------------- |
| `--ore-switch-unit`                    | 基础尺寸单位   |
| `--ore-switch-track-background`        | 未激活轨道背景 |
| `--ore-switch-track-highlight`         | 未激活轨道高光 |
| `--ore-switch-track-border`            | 轨道边框颜色   |
| `--ore-switch-thumb-background`        | 滑块背景       |
| `--ore-switch-thumb-highlight`         | 滑块高光       |
| `--ore-switch-thumb-shadow`            | 滑块阴影       |
| `--ore-switch-thumb-border`            | 滑块边框颜色   |
| `--ore-switch-active-track-background` | 激活轨道背景   |
| `--ore-switch-active-track-highlight`  | 激活轨道高光   |
| `--ore-switch-icon`                    | 滑块图标颜色   |

## 无障碍

原生 checkbox 提供表单和键盘行为，`role="switch"` 提供开关语义。使用 label 描述对应的控件。
