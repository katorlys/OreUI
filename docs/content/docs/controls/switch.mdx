    </Tab>
    <Tab value="React">

```tsx
import { Switch } from "@oreui-web/react/switch";
import { useState } from "react";

export function AutosaveSwitch() {
  const [checked, setChecked] = useState(false);

  return (
    <Switch
      checked={checked}
      name="autosave"
      value="enabled"
      variant="icons"
      onChange={(event) => {
        const control = event.target as HTMLElement & { checked: boolean };
        setChecked(control.checked);
      }}
    >
      Enable autosave
    </Switch>
  );
}
```

## Variants

```html
<ore-switch variant="default">Default</ore-switch>
<ore-switch variant="icons">Icons</ore-switch>
```

## Colors

Use `color` to change the checked state color.

```html
<ore-switch color="primary" checked>Primary</ore-switch>
<ore-switch color="secondary" checked>Secondary</ore-switch>
<ore-switch color="destructive" checked>Destructive</ore-switch>
<ore-switch color="dungeons" checked>Dungeons</ore-switch>
<ore-switch color="legends" checked>Legends</ore-switch>
<ore-switch color="realms" checked>Realms</ore-switch>
<ore-switch color="gold" checked>Gold</ore-switch>
```

## Forms

Checked values are included in `FormData`, and reset restores the initial state.

## Events

Switch dispatches native `input` and `change` events after its value changes.

## API

| Property   | Attribute  | Type                                                                                         | Default     |
| ---------- | ---------- | -------------------------------------------------------------------------------------------- | ----------- |
| `checked`  | `checked`  | `boolean`                                                                                    | `false`     |
| `color`    | `color`    | `"primary" \| "secondary" \| "destructive" \| "dungeons" \| "legends" \| "realms" \| "gold"` | `"primary"` |
| `disabled` | `disabled` | `boolean`                                                                                    | `false`     |
| `name`     | `name`     | `string`                                                                                     | `""`        |
| `required` | `required` | `boolean`                                                                                    | `false`     |
| `value`    | `value`    | `string`                                                                                     | `"on"`      |
| `variant`  | `variant`  | `"default" \| "icons"`                                                                       | `"default"` |

## Accessibility

Switch provides switch semantics and supports `Space`. Use a label to describe the content.
