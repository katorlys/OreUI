import { Textarea } from "@oreui-web/react/textarea";
import type { OreTextarea } from "oreui-web/textarea";
import { useRef, useState } from "react";

export function WorldDescription() {
  const [value, setValue] = useState("");
  const ref = useRef<OreTextarea>(null);

  return (
    <Textarea
      ref={ref}

    />
  );
}

## Resize

Textarea can be resized horizontally and vertically by default, and it can be customized through its public class.

```css
ore-textarea > .ore-textarea-control > .ore-textarea-input {
  resize: none;
}
```

## Selection and scrolling

The host proxies common native textarea APIs: `selectionStart`, `selectionEnd`, `setSelectionRange()`, `scrollTop`, and `focus()`.

```js
const textarea = document.querySelector("ore-textarea");
textarea.focus();
textarea.setSelectionRange(0, 5);
```

## Events

Textarea dispatches `input` whenever its value changes and `change` when the edit is committed. Its public `value` property stays synchronized with the native textarea.

## Public classes

| Class                       | Purpose            |
| --------------------------- | ------------------ |
| `.ore-textarea-label`       | Visible label      |
| `.ore-textarea-control`     | Textarea frame     |
| `.ore-textarea-input`       | Native textarea    |
| `.ore-textarea-description` | Supporting text    |
| `.ore-textarea-error`       | Validation message |

## CSS variables

Textarea exposes the same styling roles as Textfield with the `--ore-textarea-*` prefix, including `background`, `shadow`, `border`, `foreground`, `input-foreground`, `description`, `error`, `focus-ring`, and `caret`.

## Accessibility

Textarea connects its label, description, and validation message to the native textarea. Use a visible `label` first, and use `aria-label` only when a visible label is not appropriate.
