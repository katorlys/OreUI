import { Textfield } from "@oreui-web/react/textfield";
import { useState } from "react";

export function WorldNameField() {
  const [value, setValue] = useState("");

  return (
    <Textfield
      name="world"
      label="World name"
      description="Shown to other players"
      placeholder="New World"
      value={value}
      onInput={(event) => {
        const field = event.target as HTMLElement & { value: string };
        setValue(field.value);
      }}
    />
  );
}

## 验证

使用原生输入约束进行验证。在原生 HTML 中，可以使用 `input.setCustomValidity()` 设置自定义消息，并通过 `aria-describedby` 关联可见错误信息。

```html
<div class="ore-textfield">
  <label class="ore-textfield-label" for="seed">Seed</label>
  <span class="ore-textfield-control">
    <input
      class="ore-textfield-input"
      id="seed"
      name="seed"
      type="number"
      required
    />
  </span>
</div>
<div class="ore-textfield">
  <label class="ore-textfield-label" for="world-error">World name</label>
  <span class="ore-textfield-control">
    <input
      class="ore-textfield-input"
      id="world-error"
      name="world"
      aria-invalid="true"
      aria-describedby="world-error-message"
    />
  </span>
  <span class="ore-textfield-error" id="world-error-message"
    >This name is already in use</span
  >
</div>
```

## 事件

原生 Textfield 使用 `<input>` 自带的 `input`、`change`、`focus`、`blur` 和表单事件。值就是输入元素原生的 `value`。

React、Vue、Svelte 和 Solid 适配保持同样的原生行为，只增加适配框架的 API：React 转发`onInput` 和 `ref`，Vue 支持 `v-model`，Svelte 支持 `bind:value`，Solid 支持 `onValueChange`。

## API

原生用法没有 Textfield 自定义元素或自定义属性。请将标准输入属性直接写在原生 `<input>` 上：

| 属性                                                      | 用途                   |
| --------------------------------------------------------- | ---------------------- |
| `autocomplete`、`inputmode`、`list`                       | 浏览器输入辅助         |
| `name`、`value`、`type`                                   | 表单名称、值和输入类型 |
| `placeholder`                                             | 空值提示               |
| `min`、`max`、`step`、`minlength`、`maxlength`、`pattern` | 原生输入约束           |
| `required`、`readonly`、`disabled`                        | 原生输入状态           |
| `id`                                                      | 关联 label 和描述信息  |

框架适配接受 `label`、`description` 和 `error`，用于生成对应的 Light DOM 元素。原生 HTML 使用者需要显式写出这些元素。

## 无障碍

应尽可能为每个输入使用可见的 `<label for="...">`。`for` 必须匹配输入的 `id`，输入元素本身仍是可访问控件和表单参与者。仅当不适合显示标签时才使用 `aria-label`。

为辅助文本设置 `id`，并在输入上通过 `aria-describedby` 引用。对于可见错误信息，同样将错误元素加入该属性，并在值无效时设置 `aria-invalid="true"`。原生约束或 `setCustomValidity()` 应作为唯一的浏览器 validity 状态来源。

`disabled` 会让输入失去焦点能力并排除在表单提交之外；`readonly` 仍可聚焦和提交，但不能编辑。需要整体禁用多个字段时，请在外层使用 `<fieldset disabled>`。
