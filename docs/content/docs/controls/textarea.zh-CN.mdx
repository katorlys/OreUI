import { Textarea } from "@oreui-web/react/textarea";
import type { OreTextarea } from "oreui-web/textarea";
import { useRef, useState } from "react";

export function WorldDescription() {
  const [value, setValue] = useState("");
  const ref = useRef<OreTextarea>(null);

  return (
    <Textarea
      ref={ref}
      label="World description"

    />
  );
}

## 调整高度

Textarea 默认支持水平和垂直的大小调整，并可通过原生控件的公开类来定制该行为。

```css
ore-textarea > .ore-textarea-control > .ore-textarea-input {
  resize: none;
}
```

## 选区和滚动

宿主元素会代理常用的原生 textarea API：`selectionStart`、`selectionEnd`、`setSelectionRange()`、`scrollTop` 和 `focus()`。

```js
const textarea = document.querySelector("ore-textarea");
textarea.focus();
textarea.setSelectionRange(0, 5);
```

## 事件

Textarea 的值变化时会触发 `input`、编辑提交时会触发 `change`。公共 `value` 属性会与原生 textarea 保持同步。

## 公开类

| 类                          | 用途          |
| --------------------------- | ------------- |
| `.ore-textarea-label`       | 可见标签      |
| `.ore-textarea-control`     | 多行输入框架  |
| `.ore-textarea-input`       | 原生 textarea |
| `.ore-textarea-description` | 辅助文本      |
| `.ore-textarea-error`       | 验证信息      |

## CSS 变量

Textarea 与 Textfield 的样式相同，并使用 `--ore-textarea-*` 前缀，包含 `background`、`shadow`、`border`、`foreground`、`input-foreground`、`description`、`error`、`focus-ring` 和 `caret`。

## 无障碍

Textarea 会将标签、描述和验证信息关联到原生 textarea 之中。请优先使用可见的 `label`，只有在可见标签不适用时，才使用 `aria-label`。
