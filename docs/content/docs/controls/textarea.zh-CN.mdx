import { Textarea } from "@oreui-web/react/textarea";
import { useRef, useState } from "react";

export function WorldDescription() {
  const [value, setValue] = useState("");
  const ref = useRef<HTMLTextAreaElement>(null);

  return (
    <Textarea
      ref={ref}
      label="World description"
      rows={4}
      value={value}

  );
}

## 调整高度

Textarea 默认支持水平和垂直的大小调整，并可通过原生控件的公开类来定制该行为。

```css
.ore-textarea > .ore-textarea-control > .ore-textarea-input {
  resize: none;
}
```

## 选区和滚动

原生 `<textarea>` 直接提供 `selectionStart`、`selectionEnd`、`setSelectionRange()`、`scrollTop` 和 `focus()`。

```js
const textarea = document.querySelector("textarea");
textarea.focus();
textarea.setSelectionRange(0, 5);
```

## 事件

原生 textarea 的值变化时会触发 `input`、编辑提交时会触发 `change`。`value` 属性就是唯一的表单状态来源。

## API

| Prop                      | 属性                      | 类型                        | 默认值     |
| ------------------------- | ------------------------- | --------------------------- | ---------- |
| `autocomplete`            | `autocomplete`            | `string`                    | 浏览器默认 |
| `disabled`                | `disabled`                | `boolean`                   | `false`    |
| `inputMode`               | `inputmode`               | `string`                    | 浏览器默认 |
| `maxLength` / `minLength` | `maxlength` / `minlength` | `number`                    | 浏览器默认 |
| `name`                    | `name`                    | `string`                    | `""`       |
| `placeholder`             | `placeholder`             | `string`                    | `""`       |
| `readonly`                | `readonly`                | `boolean`                   | `false`    |
| `required`                | `required`                | `boolean`                   | `false`    |
| `rows`                    | `rows`                    | `number`                    | `2`        |
| `spellcheck`              | `spellcheck`              | `boolean`                   | 浏览器默认 |
| `wrap`                    | `wrap`                    | `"hard" \| "off" \| "soft"` | `"soft"`   |

## 公开类

| 类                          | 用途          |
| --------------------------- | ------------- |
| `.ore-textarea-label`       | 可见标签      |
| `.ore-textarea-control`     | 多行输入框架  |
| `.ore-textarea-input`       | 原生 textarea |
| `.ore-textarea-description` | 辅助文本      |
| `.ore-textarea-error`       | 验证信息      |

## CSS 变量

Textarea 与 Textfield 的样式相同，并使用 `--ore-textarea-*` 前缀，包含 `background`、`shadow`、`border`、`foreground`、`input-foreground`、`description`、`error`、`focus-ring` 和 `caret`。

## 无障碍

请使用与 `id` 匹配的可见 `label`。通过 `aria-describedby` 关联帮助或错误文本。使用原生 `required`、`minlength`、`maxlength`、`readonly` 和 `disabled` 属性表达对应状态。
