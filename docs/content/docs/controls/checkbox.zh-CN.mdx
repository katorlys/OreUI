    </Tab>
    <Tab value="React">

```tsx
import { Checkbox } from "@oreui-web/react/checkbox";
import { useState } from "react";

export function UpdatesCheckbox() {
  const [checked, setChecked] = useState(false);

  return (
    <Checkbox
      checked={checked}
      name="updates"
      value="yes"
      onChange={(event) => {
        const checkbox = event.target as HTMLElement & { checked: boolean };
        setChecked(checkbox.checked);
      }}
    >
      Receive updates
    </Checkbox>
  );
}
```

## 表单

勾选的项的值会包含在 `FormData` 中。重置表单会恢复初始的 `checked` 的状态。

```html
<form>
  <ore-checkbox name="terms" value="accepted" required>
    Accept the terms
  </ore-checkbox>
  <ore-button type="submit">Continue</ore-button>
  <ore-button type="reset" color="secondary">Reset</ore-button>
</form>
```

## 事件

Checkbox 的值发生变化后，会触发原生 `input` 和 `change` 事件。

## API

| Prop       | 属性       | 类型                                                                                         | 默认值      |
| ---------- | ---------- | -------------------------------------------------------------------------------------------- | ----------- |
| `checked`  | `checked`  | `boolean`                                                                                    | `false`     |
| `color`    | `color`    | `"primary" \| "secondary" \| "destructive" \| "dungeons" \| "legends" \| "realms" \| "gold"` | `"primary"` |
| `disabled` | `disabled` | `boolean`                                                                                    | `false`     |
| `name`     | `name`     | `string`                                                                                     | `""`        |
| `required` | `required` | `boolean`                                                                                    | `false`     |
| `value`    | `value`    | `string`                                                                                     | `"on"`      |

| 方法               | 描述                 |
| ------------------ | -------------------- |
| `checkValidity()`  | 检查当前有效性       |
| `reportValidity()` | 检查有效性并报告错误 |

## CSS 变量

| 变量                                       | 用途         |
| ------------------------------------------ | ------------ |
| `--ore-checkbox-unit`                      | 基础尺寸单位 |
| `--ore-checkbox-background`                | 未选中背景   |
| `--ore-checkbox-highlight`                 | 未选中高光   |
| `--ore-checkbox-border`                    | 边框颜色     |
| `--ore-checkbox-indicator`                 | 勾选标记颜色 |
| `--ore-checkbox-active-background`         | 选中背景     |
| `--ore-checkbox-active-background-hover`   | 选中悬停背景 |
| `--ore-checkbox-active-background-pressed` | 选中按下背景 |
| `--ore-checkbox-active-highlight`          | 选中高光     |
| `--ore-checkbox-active-highlight-hover`    | 选中悬停高光 |
| `--ore-checkbox-active-highlight-pressed`  | 选中按下高光 |

## 无障碍

Checkbox 提供相应的无障碍语义，并支持空格键。
