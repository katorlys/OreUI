## Usage

<Tabs groupId="framework" items={["Vanilla", "React", "Vue", "Svelte", "Astro"]} persist>

    </Tab>
    <Tab value="React">

```tsx
import { Slider } from "@oreui-web/react/slider";
import { useState } from "react";

export function RenderDistanceSlider() {
  const [value, setValue] = useState(12);

  return (
    <Slider
      aria-label="Render distance"
      min={2}
      max={32}
      step={1}
      value={value}
      onInput={(event) => {
        const slider = event.target as HTMLElement & { value: number };
        setValue(slider.value);
      }}
    />
  );
}
```

## Variants

```html
<ore-slider value="40" variant="default" aria-label="Volume"></ore-slider>
<ore-slider value="40" variant="segmented" aria-label="Volume"></ore-slider>
```

## Events

Slider dispatches `input` while its value changes and `change` when the interaction is committed.

## API

| Property      | Attribute     | Type                                                                                         | Default        |
| ------------- | ------------- | -------------------------------------------------------------------------------------------- | -------------- |
| `color`       | `color`       | `"primary" \| "secondary" \| "destructive" \| "dungeons" \| "legends" \| "realms" \| "gold"` | `"primary"`    |
| `disabled`    | `disabled`    | `boolean`                                                                                    | `false`        |
| `max`         | `max`         | `number`                                                                                     | `100`          |
| `min`         | `min`         | `number`                                                                                     | `0`            |
| `name`        | `name`        | `string`                                                                                     | `""`           |
| `orientation` | `orientation` | `"horizontal" \| "vertical"`                                                                 | `"horizontal"` |
| `step`        | `step`        | `number`                                                                                     | `1`            |
| `value`       | `value`       | `number`                                                                                     | `0`            |
| `variant`     | `variant`     | `"default" \| "segmented"`                                                                   | `"default"`    |

## CSS variables

| Variable                               | Purpose                   |
| -------------------------------------- | ------------------------- |
| `--ore-slider-unit`                    | Base size unit            |
| `--ore-slider-track-border`            | Track border color        |
| `--ore-slider-active-range-background` | Active range background   |
| `--ore-slider-active-range-highlight`  | Active range highlight    |
| `--ore-slider-track-background`        | Inactive track background |
| `--ore-slider-track-highlight`         | Inactive track highlight  |
| `--ore-slider-thumb-background`        | Thumb background          |

## Accessibility

Add an accessible name with `aria-label`. Slider uses a native range input for pointer and keyboard interaction.
