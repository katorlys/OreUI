import { Select } from "@oreui-web/react/select";
import { useState } from "react";

export function GameModeSelect() {
  const [value, setValue] = useState("creative");

  return (
    <Select
      value={value}
      onChange={(event) => setValue(event.currentTarget.value)}
    >
      <option value="survival">Survival</option>
      <option value="creative">Creative</option>
    </Select>
  );
}

## Native behavior

The closed control uses the same OreUI surface and icon as the Dropdown trigger. The opened option panel is rendered by the browser and operating system, so its exact appearance is platform-dependent.

Use Select for plain text options. Use [Dropdown](./dropdown) only when options require custom content or a fully styled popup.

## Forms and events

Select uses the native `HTMLSelectElement` API. `disabled`, `required`, `name`, `value`, validation, `FormData`, keyboard navigation, and form reset remain browser behavior. Listen for the standard `input` or `change` event.
