import { OreButton } from "@katorlys/oreui-react/button";
import { OreDropdown } from "@katorlys/oreui-react/dropdown";
import { useState } from "react";

export function GameModeDropdown() {
	const [value, setValue] = useState("creative");

	return (
		<OreDropdown value={value} variant="bordered" onChange={(event) => setValue(event.detail.value)}>
			<OreButton className="ore-dropdown-trigger" type="button" variant="secondary">
				<span className="ore-dropdown-trigger-label">Game mode</span>
			</OreButton>
			<div className="ore-dropdown-menu">

				<button className="ore-dropdown-item" type="button" data-value="creative">Creative</button>
				<button className="ore-dropdown-item" type="button" data-value="adventure">Adventure</button>
			</div>
		</OreDropdown>
	);
}

## Variants
`borderless` shows item borders on interaction, while `bordered` keeps them visible.

## Events
|Event|Detail|
|---|---|
|`change`|`{ item: HTMLElement; value: string }`|
|`open-change`|`boolean`|

In React, use `onChange` and `onOpenChange`.

## API
|Property|Attribute|Type|Default|
|---|---|---|---|
|`defaultOpen`|`default-open`|`boolean`|`false`|
|`open`|`open`|`boolean`|`false`|
|`value`|`value`|`string`|`""`|
|`variant`|`variant`|`"bordered" \| "borderless"`|`"borderless"`|

## Public classes
|Class|Purpose|
|---|---|
|`.ore-dropdown-trigger`|Button that opens the menu|
|`.ore-dropdown-trigger-label`|Text updated from the selected item|
|`.ore-dropdown-menu`|Popover menu|
|`.ore-dropdown-item`|Selectable menu item|

## Accessibility
Dropdown manages menu roles, selection state, focus, and `Arrow`, `Home`, `End`, `Enter`, `Space`, `Escape`, and `Tab` keys.  
Use native `disabled` on unavailable items.
